# TEST 2: Operator Precedence - ^ (multiply) binds tighter than +/-
# z = a + b ^ c - d / e % f
# Expected quads (^ and / and % before + and -):
#   1 | ^  | b | c  | t1      <- b^c first
#   2 | /  | d | e  | t2      <- d/e

#   4 | +  | a | t1 | t4      <- a + t1
#   5 | -  | t4| t3 | t5      <- t4 - t3
#   6 | =  | t5| -  | z

# TEST 3: Parentheses Override Precedence
# Without parens: a + b^c  -> t1=b^c, t2=a+t1   (multiply first)
# With parens:  (a + b)^c  -> t1=a+b, t2=t1^c   (addition first)
# Expected quads:
#   1 | +  | a  | b  | t1     <- addition FIRST (parenthesised)
#   2 | ^  | t1 | c  | t2     <- then multiply
#   3 | =  | t2 | -  | x
#   4 | ^  | b  | c  | t3     <- multiply FIRST (no parens)
#   5 | +  | a  | t3 | t4     <- then addition
#   6 | =  | t4 | -  | y

int main() begin
    int a;
    int b;
    int c;
    int x;
    int y;
